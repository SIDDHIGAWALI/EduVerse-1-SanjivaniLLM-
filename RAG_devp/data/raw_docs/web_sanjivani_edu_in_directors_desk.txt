// Handle clicking on the search icon to show/hide the input

document.querySelector('.search-icon').addEventListener('click', function(event) {

const searchInput = document.querySelector('.search-input');

searchInput.style.display = 'block'; // Show the input field

event.stopPropagation(); // Prevent the click from propagating to the document

});

// Handle clicking outside the input to hide it

document.addEventListener('click', function(event) {

const searchInput = document.querySelector('.search-input');

const searchIcon = document.querySelector('.search-icon');

// Check if the click is outside the input or the icon

if (!searchInput.contains(event.target) && !searchIcon.contains(event.target)) {

searchInput.style.display = 'none'; // Hide the input field

}

});

// Prevent hiding the input when clicking inside it

document.querySelector('.search-input').addEventListener('click', function(event) {

event.stopPropagation(); // Prevents the document click event from firing

});
